Examples

The code on this page is run when the documentation is built. The schemas are in the examples/ directory of the repository.

A product catalog

This schema is from the Getting Started Step-By-Step guide of the JSON Schema organization. It describes a product with a nested object and a few constraints:

schema = joinpath(EXAMPLES, "acme_schemas", "acme_nestedschema.json")
print(read(schema, String))
{
    "$schema": "https://json-schema.org/draft/2020-12/schema",
    "$id": "https://example.com/product.schema.json",
    "title": "Product",
    "description": "A product from Acme's catalog",
    "type": "object",
    "properties": {
      "productId": {
        "description": "The unique identifier for a product",
        "type": "integer"
      },
      "productName": {
        "description": "Name of the product",
        "type": "string"
      },
      "price": {
        "description": "The price of the product",
        "type": "number",
        "exclusiveMinimum": 0
      },
      "tags": {
        "description": "Tags for the product",
        "type": "array",
        "items": {
          "type": "string"
        },
        "minItems": 1,
        "uniqueItems": true
      },
      "dimensions": {
        "description": "Dimensions of a product",
        "type": "object",
        "properties": {
          "length": {
            "type": "number"
          },
          "width": {
            "type": "number"
          },
          "height": {
            "type": "number"
          }
        },
        "required": [ "length", "width", "height" ]
      }
    },
    "required": [ "productId", "productName", "price" ]
  }

Generate a module from it, and load it:

file = generate_module(schema, joinpath(OUT, "ProductTypes.jl"))
include(file)
using .ProductTypes

These are the generated types (the rest of the file is the code for reading, writing and validating, the same in every generated module):

show_types(file)
# ---------------------------------------------------------------------------------------

# ---------------------------------------------------------------------------------------
# Types from `acme_nestedschema.json`
# ---------------------------------------------------------------------------------------

"""
    ProductDimensions(; kwargs...)

Dimensions of a product

# Fields

- `length::Float64` (required)
- `width::Float64` (required)
- `height::Float64` (required)
- `_extra::Dict{String, Any}`: additional properties, not described in the schema
"""
struct ProductDimensions <: AbstractSchemaObject
    length::Float64
    width::Float64
    height::Float64
    _extra::Dict{String, Any}
    function ProductDimensions(; length, width, height, _extra = Dict{String, Any}())
        length = _jsfrom(Float64, length)
        width = _jsfrom(Float64, width)
        height = _jsfrom(Float64, height)
        _extra = _jsfrom(Dict{String, Any}, _extra)
        _jscheck(ProductDimensions, (length, width, height), _extra)
        return new(length, width, height, _extra)
    end
end
_SPECS[ProductDimensions] = (
    fields = (
        (name = :length, key = "length", required = true, nullable = false, constraints = nothing),
        (name = :width, key = "width", required = true, nullable = false, constraints = nothing),
        (name = :height, key = "height", required = true, nullable = false, constraints = nothing),
    ),
    additionalProperties = true,
    extra = :_extra,
    minProperties = nothing,
    maxProperties = nothing,
    schema = nothing,
)

"""
    Product(; kwargs...)

A product from Acme's catalog

# Fields

- `productId::Int64` (required): The unique identifier for a product
- `productName::String` (required): Name of the product
- `price::Float64` (required): The price of the product
- `tags::Union{Nothing, Vector{String}}`: Tags for the product
- `dimensions::Union{Nothing, ProductDimensions}`: Dimensions of a product
- `_extra::Dict{String, Any}`: additional properties, not described in the schema
"""
struct Product <: AbstractSchemaObject
    productId::Int64
    productName::String
    price::Float64
    tags::Union{Nothing, Vector{String}}
    dimensions::Union{Nothing, ProductDimensions}
    _extra::Dict{String, Any}
    function Product(; productId, productName, price, tags = nothing, dimensions = nothing, _extra = Dict{String, Any}())
        productId = _jsfrom(Int64, productId)
        productName = _jsfrom(String, productName)
        price = _jsfrom(Float64, price)
        tags = _jsfrom(Union{Nothing, Vector{String}}, tags)
        dimensions = _jsfrom(Union{Nothing, ProductDimensions}, dimensions)
        _extra = _jsfrom(Dict{String, Any}, _extra)
        _jscheck(Product, (productId, productName, price, tags, dimensions), _extra)
        return new(productId, productName, price, tags, dimensions, _extra)
    end
end
_SPECS[Product] = (
    fields = (
        (name = :productId, key = "productId", required = true, nullable = false, constraints = nothing),
        (name = :productName, key = "productName", required = true, nullable = false, constraints = nothing),
        (name = :price, key = "price", required = true, nullable = false, constraints = (exclusiveMinimum = 0,)),
        (name = :tags, key = "tags", required = false, nullable = false, constraints = (minItems = 1, uniqueItems = true,)),
        (name = :dimensions, key = "dimensions", required = false, nullable = false, constraints = nothing),
    ),
    additionalProperties = true,
    extra = :_extra,
    minProperties = nothing,
    maxProperties = nothing,
    schema = nothing,
)


"""
    RootType

Alias of `Product`, the type of the root of the schema.
"""
const RootType = Product

end # module ProductTypes

Values are built with keyword constructors. Plain values are converted, and optional fields can be left out:

p = Product(productId = 1, productName = "An ice sculpture", price = 12.5,
            dimensions = Dict("length" => 7, "width" => 12, "height" => 9.5))
Main.ProductTypes.Product(1, "An ice sculpture", 12.5, nothing, Main.ProductTypes.ProductDimensions(7.0, 12.0, 9.5, Dict{String, Any}()), Dict{String, Any}())
p.dimensions
Main.ProductTypes.ProductDimensions(7.0, 12.0, 9.5, Dict{String, Any}())

and are written as JSON, without the fields that are not set:

print(tojson(p; pretty = true))
{
  "productId": 1,
  "productName": "An ice sculpture",
  "price": 12.5,
  "dimensions": {
    "length": 7,
    "width": 12,
    "height": 9.5
  }
}

Reading a document validates it against the schema:

json = """{"productId": 1, "productName": "An ice sculpture", "price": 0}"""
try
    fromjson(Product, json)
catch err
    showerror(stdout, err)
end
SchemaValidationError: `price` in `Product` must be > 0, got 0.0
json = """{"productId": 1, "productName": "Ice", "price": 1, "tags": ["cold", "cold"]}"""
try
    fromjson(Product, json)
catch err
    showerror(stdout, err)
end
SchemaValidationError: `tags` in `Product` must have unique items

A schema referenced by its $id

Another version of the product schema refers to a separate schema for the location of the warehouse by its $id, https://example.com/geographical-location.schema.json. The file with that schema is given with the schemas keyword:

file = generate_module(joinpath(EXAMPLES, "acme_schemas", "acme_withreferenceschema.json"),
    joinpath(OUT, "WarehouseTypes.jl"); module_name = "WarehouseTypes",
    schemas = [joinpath(EXAMPLES, "acme_schemas", "acme_tobereferredschema.json")])
include(file)
using .WarehouseTypes: Product as WarehouseProduct, LongitudeAndLatitude
data = joinpath(EXAMPLES, "acme_schemas", "acme_data.json")
print(read(data, String))
{
    "productId": 1,
    "productName": "An ice sculpture",
    "price": 12.50,
    "tags": [ "cold", "ice" ],
    "dimensions": {
        "length": 7.0,
        "width": 12.0,
        "height": 9.5
    },
    "warehouseLocation": {
        "latitude": -78.75,
        "longitude": 20.4
    }
}
product = WarehouseTypes.readjson(WarehouseProduct, data)
product.warehouseLocation
Main.WarehouseTypes.LongitudeAndLatitude(-78.75, 20.4, Dict{String, Any}())

The referenced schema has the title "Longitude and Latitude", which gives the name of its type, LongitudeAndLatitude.

oneOf, anyOf, allOf and not

This schema shows the keywords for combining schemas:

schema = joinpath(EXAMPLES, "more_schemas", "combined_schema.json")
print(read(schema, String))
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "CombinedSchemaExample",
  "description": "A single schema demonstrating oneOf, anyOf, allOf, and not.",
  "type": "object",
  "properties": {
    "one_of_user": {
      "description": "Demonstrates 'oneOf' for a user that is either a Person or a Company.",
      "oneOf": [
        {
          "title": "Person",
          "type": "object",
          "properties": {
            "user_type": {
              "type": "string",
              "const": "person"
            },
            "first_name": {
              "type": "string"
            },
            "last_name": {
              "type": "string"
            }
          },
          "required": ["user_type", "first_name", "last_name"]
        },
        {
          "title": "Company",
          "type": "object",
          "properties": {
            "user_type": {
              "type": "string",
              "const": "company"
            },
            "company_name": {
              "type": "string"
            }
          },
          "required": ["user_type", "company_name"]
        }
      ]
    },
    "any_of_identifier": {
      "description": "Demonstrates 'anyOf' for a product identifier.",
      "anyOf": [
        {
          "type": "object",
          "properties": {
            "sku": {
              "type": "string"
            }
          },
          "required": ["sku"]
        },
        {
          "type": "object",
          "properties": {
            "product_code": {
              "type": "string"
            }
          },
          "required": ["product_code"]
        }
      ]
    },
    "all_of_request": {
      "description": "Demonstrates 'allOf' to combine schemas for a request with authentication.",
      "allOf": [
        {
          "type": "object",
          "properties": {
            "payload": {
              "type": "object"
            }
          },
          "required": ["payload"]
        },
        {
          "type": "object",
          "properties": {
            "api_key": {
              "type": "string"
            },
            "timestamp": {
              "type": "string"
            }
          },
          "required": ["api_key", "timestamp"]
        }
      ]
    },
    "not_a_password": {
      "description": "A password that cannot be the literal string 'password'.",
      "type": "string",
      "not": {
        "const": "password"
      }
    }
  },
  "required": [
    "one_of_user",
    "any_of_identifier",
    "all_of_request",
    "not_a_password"
  ]
}
file = generate_module(schema, joinpath(OUT, "CombinedTypes.jl"); module_name = "CombinedTypes")
include(file)
using .CombinedTypes
fieldtypes(CombinedSchemaExample)
(Union{Main.CombinedTypes.Company, Main.CombinedTypes.Person}, Union{Main.CombinedTypes.CombinedSchemaExampleAnyOfIdentifierOption1, Main.CombinedTypes.CombinedSchemaExampleAnyOfIdentifierOption2}, Main.CombinedTypes.CombinedSchemaExampleAllOfRequest, String, Dict{String, Any})

The options of oneOf have titles, so they are the types Person and Company. The options of anyOf have none, so their names are made from the parent and the property. The allOf is merged into a single type, and the not is checked at runtime. When reading, the option that fits is chosen:

json = """
{
  "one_of_user": {"user_type": "company", "company_name": "Acme"},
  "any_of_identifier": {"sku": "A-1"},
  "all_of_request": {"payload": {}, "api_key": "k", "timestamp": "2026-11-13"},
  "not_a_password": "s3cret"
}
"""
x = fromjson(RootType, json)
x.one_of_user
Main.CombinedTypes.Company(CompanyUserTypeEnum("company"), "Acme", Dict{String, Any}())

and the not is enforced:

try
    fromjson(RootType, replace(json, "s3cret" => "password"))
catch err
    showerror(stdout, err)
end
SchemaValidationError: `not_a_password` in `CombinedSchemaExample` is invalid: must not match the schema in `not`
